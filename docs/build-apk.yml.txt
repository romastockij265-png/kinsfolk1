name: Build APK
on:
  workflow_dispatch:
permissions:
  contents: write
jobs:
  apk:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - name: Prepare web files and Capacitor project
        run: |
          set -e
          VER=$(grep -o "[0-9]*\.[0-9]*\.[0-9]*" js/core/version.js | head -1)
          echo "VER=$VER" >> $GITHUB_ENV
          mkdir -p build/www
          cp -r index.html css js build/www/
          [ -d fonts ] && cp -r fonts build/www/ || true
          cd build
          npm init -y > /dev/null
          npm install @capacitor/core@6.2.0 @capacitor/cli@6.2.0 @capacitor/android@6.2.0
          cat > capacitor.config.json <<'EOF'
          {
            "appId": "com.kinsfolk.app",
            "appName": "Kinsfolk",
            "webDir": "www",
            "backgroundColor": "#10141D"
          }
          EOF
          npx cap add android
      - name: Icons, permissions, version
        run: |
          set -e
          cd build
          cp -r ../assets/android/res/* android/app/src/main/res/
          sed -i 's#@drawable/splash#@color/ic_launcher_background#' android/app/src/main/res/values/styles.xml
          python3 - <<'PY'
          p='android/app/src/main/AndroidManifest.xml'
          s=open(p).read()
          perms=''.join('    <uses-permission android:name="android.permission.%s" />\n' % x for x in ['CAMERA','RECORD_AUDIO','MODIFY_AUDIO_SETTINGS','ACCESS_NETWORK_STATE'])
          s=s.replace('</manifest>', perms + '</manifest>')
          open(p,'w').write(s)
          PY
          sed -i "s/versionCode 1/versionCode ${{ github.run_number }}/; s/versionName \"1.0\"/versionName \"$VER\"/" android/app/build.gradle
          npx cap sync android
      - name: Build APK
        run: |
          set -e
          mkdir -p ~/.android
          cp assets/debug.keystore ~/.android/debug.keystore
          cd build/android
          chmod +x gradlew
          ./gradlew assembleDebug --no-daemon
          cp app/build/outputs/apk/debug/app-debug.apk ../../Kinsfolk.apk
      - uses: actions/upload-artifact@v4
        with:
          name: Kinsfolk-apk
          path: Kinsfolk.apk
      - name: Publish release
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          gh release create "apk-${{ github.run_number }}" Kinsfolk.apk --title "Kinsfolk $VER (сборка ${{ github.run_number }})" --notes "Скачайте Kinsfolk.apk и установите на телефон." --latest
